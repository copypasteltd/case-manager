func TestDOCXTextExtraction(t *testing.T) {
	ctx := context.Background()
	t.Run("referenced stories revisions tables controls fields and textboxes", func(t *testing.T) {
		body := officeZip(t, complexWordParts(), 0)
		got, err := ExtractText(ctx, docxMediaType, body)
		if err != nil {
			t.Fatal(err)
		}
		for _, want := range []string{"香港🙂 合同", "[插入]新增条款[/插入]", "[删除]删除条款[/删除]", "[移出]移动原文[/移出]", "[移入]移动新文[/移入]", "链接显示文字", "[表格]", "[单元格]HKD 1200", "[直接隐藏文字]隐藏条款", "[文本框]", "文本框交付说明", "字段指令，不执行：DATE", "缓存结果，可能过期：旧日期缓存", "INCLUDETEXT https://example.invalid/private", "外部字段旧缓存", "第 1 节页眉 default", "公司页眉", "第 1 节页脚 first", "合同页脚", "[脚注 2]", "脚注付款依据", "[尾注 3]", "尾注有效期", "[批注 5]", "批注需财务确认"} {
			if !strings.Contains(got, want) {
				t.Errorf("missing %q in %s", want, got)
			}
		}
		if strings.Contains(got, "DO_NOT_INDEX") || strings.Count(got, "脚注付款依据") != 1 || strings.Count(got, "批注需财务确认") != 1 {
			t.Fatal("unreferenced, duplicate or metadata content indexed", got)
		}
		if TextExtractorFor(strings.ToUpper(docxMediaType)+"; charset=UTF-8") != DOCXTextExtractor {
			t.Fatal("wrong selector")
		}
	})
	t.Run("strict namespace", func(t *testing.T) {
		p := complexWordParts()
		for name, value := range p {
			p[name] = strings.ReplaceAll(strings.ReplaceAll(value, wordNS, "http://purl.oclc.org/ooxml/wordprocessingml/main"), relNS, "http://purl.oclc.org/ooxml/officeDocument/relationships")
		}
		if got, err := ExtractText(ctx, docxMediaType, officeZip(t, p, 0)); err != nil || !strings.Contains(got, "脚注付款依据") {
			t.Fatal(got, err)
		}
	})
	t.Run("legacy queued algorithm remains original", func(t *testing.T) {
		body := docxFixture(t, `<w:document xmlns:w="`+wordNS+`"><w:body><w:p><w:r><w:t>旧正文</w:t></w:r><w:del><w:r><w:delText>删除旧文</w:delText></w:r></w:del></w:p></w:body></w:document>`, nil)
		if got, err := extractTextWith(ctx, docxMediaType, body, TextExtractor); err != nil || got != "旧正文" {
			t.Fatal(got, err)
		}
		if _, err := ExtractText(ctx, docxMediaType, body); err == nil {
			t.Fatal("new extraction accepted missing package relationship")
		}
	})
	for _, tc := range []struct {
		name   string
		change func(map[string]string)
	}{
		{"missing note", func(p map[string]string) { delete(p, "custom/stories/foot.xml") }},
		{"unresolved note ID", func(p map[string]string) {
			p["custom/main.xml"] = strings.ReplaceAll(p["custom/main.xml"], `w:id="2"`, `w:id="8"`)
		}},
		{"duplicate note ID", func(p map[string]string) {
			p["custom/stories/foot.xml"] = strings.ReplaceAll(p["custom/stories/foot.xml"], `w:id="99"`, `w:id="2"`)
		}},
		{"separator reference", func(p map[string]string) {
			p["custom/stories/foot.xml"] = strings.Replace(p["custom/stories/foot.xml"], `w:id="2"`, `w:id="2" w:type="separator"`, 1)
		}},
		{"external header", func(p map[string]string) {
			p["custom/_rels/main.xml.rels"] = strings.Replace(p["custom/_rels/main.xml.rels"], `Target="stories/top.xml"`, `Target="https://example.invalid/private" TargetMode="External"`, 1)
		}},
		{"wrong header kind", func(p map[string]string) { p["custom/stories/top.xml"] = p["custom/stories/bottom.xml"] }},
		{"missing relationship", func(p map[string]string) {
			p["custom/main.xml"] = strings.Replace(p["custom/main.xml"], `r:id="header"`, `r:id="absent"`, 1)
		}},
		{"ambiguous note relationship", func(p map[string]string) {
			p["custom/_rels/main.xml.rels"] = strings.Replace(p["custom/_rels/main.xml.rels"], `/comments"`, `/footnotes"`, 1)
		}},
		{"malformed note", func(p map[string]string) { p["custom/stories/comments.xml"] = "<broken>" }},
		{"DTD", func(p map[string]string) { p["custom/stories/top.xml"] = "<!DOCTYPE x>" + p["custom/stories/top.xml"] }},
		{"unclosed field", func(p map[string]string) {
			p["custom/main.xml"] = strings.Replace(p["custom/main.xml"], `<w:fldChar w:fldCharType="end"/>`, "", 1)
		}},
		{"field double separator", func(p map[string]string) {
			p["custom/main.xml"] = strings.Replace(p["custom/main.xml"], `<w:fldChar w:fldCharType="end"/>`, `<w:fldChar w:fldCharType="separate"/>`, 1)
		}},
		{"conditional alternate content", func(p map[string]string) {
			p["custom/stories/top.xml"] = `<mc:AlternateContent xmlns:mc="http://schemas.openxmlformats.org/markup-compatibility/2006"/>`
		}},
		{"external subdocument", func(p map[string]string) {
			p["custom/main.xml"] = strings.Replace(p["custom/main.xml"], "<w:body>", "<w:body><w:altChunk/>", 1)
		}},
		{"macro", func(p map[string]string) { p["word/vbaProject.bin"] = "macro" }},
		{"path escape", func(p map[string]string) { p["../escape"] = "x" }},
		{"text output bound", func(p map[string]string) {
			p["custom/stories/top.xml"] = strings.Replace(p["custom/stories/top.xml"], "公司页眉", strings.Repeat("x", TextOutputMaxBytes+1), 1)
		}},
	} {
		t.Run(tc.name, func(t *testing.T) {
			p := complexWordParts()
			tc.change(p)
			if got, err := ExtractText(ctx, docxMediaType, officeZip(t, p, 0)); err == nil || got != "" {
				t.Fatal("partial/unsafe extraction", got, err)
			}
		})
	}
}
